'; } } function updateDashboardGraphs() { const taskCanvas = $('productivityChart'); const financeCanvas = $('financeChart'); if (taskCanvas) drawProductivityGraph(taskCanvas); if (financeCanvas) drawFinanceGraph(financeCanvas); } function drawProductivityGraph(canvas) { const ctx = canvas.getContext('2d'); if (!ctx) return; const width = canvas.clientWidth || 320; const height = 190; canvas.width = width * devicePixelRatio; const key = [ date.getFullYear(), String(date.getMonth() + 1).padStart(2, '0'), String(date.getDate()).padStart(2, '0') ].join('-'); days.push({ label: date.toLocaleDateString(undefined, { weekday: 'short' }), count: db.tasks.filter(task => task.completed && (task.completedAt || task.date || '').startsWith(key) ).length }); } const max = Math.max(1, ...days.map(day => day.count)); const chartHeight = 125; const bottom = 150; const gap = width / days.length; const barWidth = Math.min(24, gap * 0.52); days.forEach((day, index) => { const x = gap * index + (gap - barWidth) / 2; const barHeight = day.count / max * chartHeight; ctx.fillStyle = '#f3a6c8'; ctx.beginPath(); ctx.roundRect( x, bottom - barHeight, barWidth, barHeight, 6 ); ctx.fill(); ctx.fillStyle = '#80536a'; ctx.font = '11px sans-serif'; ctx.textAlign = 'center'; ctx.fillText(day.label, x + barWidth / 2, 172); ctx.fillText( String(day.count), x + barWidth / 2, bottom - barHeight - 7 ); }); } function drawFinanceGraph(canvas) { const ctx = canvas.getContext('2d'); if (!ctx) return; const width = canvas.clientWidth || 320; const height = 190; canvas.width = width * devicePixelRatio; canvas.height = height * devicePixelRatio; ctx.scale(devicePixelRatio, devicePixelRatio); ctx.clearRect(0, 0, width, height); const now = new Date(); const year = now.getFullYear(); const month = now.getMonth(); const income = db.expenses.filter(item => item.type === 'income' && item.date && item.date.startsWith( `${year}-${String(month + 1).padStart(2, '0')}` ) ).reduce((sum, item) => sum + Number(item.amount || 0), 0); const spending = db.expenses.filter(item => item.type !== 'income' && item.date && item.date.startsWith( `${year}-${String(month + 1).padStart(2, '0')}` ) ).reduce((sum, item) => sum + Number(item.amount || 0), 0); const max = Math.max(1, income, spending); const chartHeight = 115; const bottom = 145; const barWidth = Math.min(55, width / 4); [ { label: 'Income', amount: income, color: '#a9dfbf' }, { label: 'Spent', amount: spending, color: '#f3a6c8' } ].forEach((item, index) => { const x = width * (index === 0 ? 0.3 : 0.7) - barWidth / 2;
